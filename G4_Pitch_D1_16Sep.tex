% ================== 3b. LUONG DIEU KHIEN ==================
\begin{frame}{Luồng điều khiển (Control Flow) [chi tiết]}
\footnotesize
\begin{enumerate}
  \item \textbf{Tiếp nhận \& Phân rã mục tiêu}: Orchestrator tra cứu ngữ cảnh phiên làm việc và phân rã yêu cầu tuân thủ phức tạp thành các câu hỏi con.
  \item \textbf{Điều phối tác tử chuyên trách}: Giao việc cho sub-agent phù hợp --- LawGraph Agent (duyệt chọn lọc đồ thị quan hệ) hoặc Legal Update Agent (xác minh hiệu lực thời gian thực).
  \item \textbf{Dự thảo \& Thẩm định chéo}: Compliance Drafter soạn thảo hồ sơ; Claim Auditor phân rã từng tuyên bố (claim decomposition) để đối soát nghiêm ngặt với điều khoản nguồn.
  \item \textbf{Vòng lặp tinh chỉnh (Refinement Loop)}: Nếu phát hiện mệnh đề thiếu căn cứ, Orchestrator kích hoạt truy xuất bổ sung (giới hạn tối đa 3 chu kỳ).
  \item \textbf{Phê duyệt \& Kết xuất}: Khi 100\% mệnh đề có căn cứ $\rightarrow$ qua HUMAN GATE (kiểm soát các tác vụ ghi không thể đảo nghịch) $\rightarrow$ kết xuất hồ sơ tuân thủ, lưu audit log và cập nhật memory dài hạn.
\end{enumerate}
\vskip0.15cm
{\scriptsize Orchestrator đóng vai trò \textbf{Supervisor}: duy trì trạng thái chung, điều phối các sub-agent độc lập và quản lý điều kiện dừng. Các sub-agent \textbf{không} gọi chéo trực tiếp mà thông qua bus điều phối của Orchestrator.}
\end{frame}

% ================== 4. CO CHE LOI: RETRIEVAL DECISION ==================
\begin{frame}{Cơ chế lõi: Quyết định duyệt đồ thị chọn lọc [chi tiết]}
{\small Cơ chế duyệt đồ thị chọn lọc (Selective Traversal) giúp tác tử tập trung vào các điều khoản liên quan trực tiếp, tránh nạp dư thừa làm tràn ngữ cảnh:}
\vskip0.2cm
\begin{columns}[T]
\begin{column}{0.32\textwidth}
\begin{block}{\small 1. Khám phá metadata quan hệ}
\footnotesize Với mỗi điều luật khởi tạo, tác tử chỉ trích xuất metadata của các cạnh liên kết: loại quan hệ, số hiệu văn bản đích, trích yếu tiêu đề; chưa nạp toàn văn để tiết kiệm context.
\end{block}
\end{column}
\begin{column}{0.32\textwidth}
\begin{block}{\small 2. Đánh giá \& Xếp hạng cạnh}
\footnotesize Xếp hạng mức độ ưu tiên theo hai tiêu chí: trọng số loại quan hệ (\textbf{Sửa đổi} $>$ \textbf{Bãi bỏ} $>$ \textbf{Thay thế} $>$ \textbf{Hướng dẫn}) và độ tương đồng ngữ nghĩa với truy vấn con.
\end{block}
\end{column}
\begin{column}{0.32\textwidth}
\begin{block}{\small 3. Duyệt chọn lọc có điểm dừng}
\footnotesize Chỉ duyệt theo 1--2 cạnh có điểm cao nhất và nạp toàn văn điều luật đích. Dừng khi Auditor xác nhận đủ căn cứ pháp lý, hoặc đạt độ sâu tối đa (2 hops). Chủ động hỏi người dùng khi thiếu dữ kiện.
\end{block}
\end{column}
\end{columns}
\vskip0.15cm
{\footnotesize \textcolor{hcmutblue}{\textbf{Thách thức kỹ thuật cốt lõi:}} Cân bằng tối ưu giữa việc bảo đảm đầy đủ căn cứ pháp lý và kiểm soát nghiêm ngặt độ dài ngữ cảnh (context bloat).}
\end{frame}

% ==================== 6b. HE CHI SO DO LUONG (METRICS) ====================
\begin{frame}{Hệ chỉ số đo lường thực nghiệm (Evaluation Metrics) [chi tiết]}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{\small 1. Năng lực truy xuất căn cứ (Retrieval)}
\footnotesize
\begin{itemize}
  \item \textbf{Precision@2}: Đo lường độ tinh lọc, loại bỏ nhiễu ngữ cảnh (không nhồi điều khoản rác vào prompt). Khắc phục điểm yếu tụt Precision@2 (0.37--0.39) của baseline 1-hop tĩnh.
  \item \textbf{Recall}: Đo lường mức độ bao phủ, không bỏ sót các điều khoản bắt buộc và thông tư hướng dẫn kèm theo.
  \item \textbf{F2-Score}:
    $$F_2 = 5 \times \frac{\text{Precision} \times \text{Recall}}{4\,\text{Precision} + \text{Recall}}$$
    Ưu tiên Recall vì trong tư vấn pháp lý, \textit{bỏ sót căn cứ luật nguy hiểm hơn thừa thông tin}.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
